## 使用示例

### 基础用法
```old8
import "firstui" as ui

app <- ui.App()

MainView() -> {
    return ui.Column({
        children: {
            ui.Text("欢迎使用 FirstUI"),
                .fontSize(24),
                .fontWeight("bold"),
            
            ui.Button("点击我")
                .onClick(() -> {
                    ui.ShowToast("Hello World!")
                })
        }
    })
}

app.Run(() -> MainView())
```

### 高级用法
```old8
// 使用新组件
ui.Select()
    .addOption("选项1", "option1")
    .addOption("选项2", "option2")
    .onChanged((value) -> {
        PrintLine("选择了: " + value)
    })

ui.NavigationView()
    .addMenuItem("首页", "home", "🏠")
    .addMenuItem("设置", "settings", "⚙️")
    .onSelectionChanged((value) -> {
        PrintLine("导航到: " + value)
    })

ui.Panel()
    .setTitle("面板标题", "副标题")
    .setContent(ui.Text("面板内容"))
```
