## 📝 使用示例

### 基础应用示例
```old8
import "firstui" as ui

app <- ui.App()

MainView() -> {
    return ui.Column({
        children: {
            ui.Text("欢迎使用 Old8Lang FirstUI")
                .fontSize(24)
                .fontWeight("bold")
                .color("#333"),
            
            ui.Row({
                children: {
                    ui.Button("点击我")
                        .onClick(() -> {
                            ui.ShowToast("Hello FirstUI!", 3000)
                        }),
                    
                    ui.Switch(true)
                        .onChanged((value) -> {
                            PrintLine("开关状态: " + value.ToStr())
                        })
                }
            }).spacing(20),
            
            ui.Table()
                .addColumn("姓名", "Name")
                .addColumn("年龄", "Age")
                .addRow({"张三", "25"})
                .addRow({"李四", "30"})
        }
    }).padding(20)
}

app.Run(() -> MainView())
```
