# 使用ControlFlowManager优化控制流管理

## 1. 设计思路

- 将控制流管理从VariateManager中分离出来，创建专门的ControlFlowManager类
- 符合单一职责原则，模块化设计
- 使用栈结构管理嵌套循环的控制流状态
- 在每次循环迭代开始时重置控制流标志

### 6.1 ControlFlowManager类设计

```csharp
public class ControlFlowManager
{
    private readonly Stack<ControlFlowState> _controlFlowStack = [];
    
    public void PushState()
    {
        _controlFlowStack.Push(new ControlFlowState());
    }
    
    public void PopState()
    {
        if (_controlFlowStack.Count > 0)
        {
            _controlFlowStack.Pop();
        }
    }
    
    public void ResetCurrentState()
    {
        if (_controlFlowStack.Count > 0)
        {
            var current = _controlFlowStack.Pop();
            _controlFlowStack.Push(new ControlFlowState());
        }
    }
    
    public bool BreakFlag
    {
        get => _controlFlowStack.Count > 0 ? _controlFlowStack.Peek().BreakFlag : false;
        set
        {
            if (_controlFlowStack.Count > 0)
            {
                _controlFlowStack.Peek().BreakFlag = value;
            }
        }
    }
    
    public bool ContinueFlag
    {
        get => _controlFlowStack.Count > 0 ? _controlFlowStack.Peek().ContinueFlag : false;
        set
        {
            if (_controlFlowStack.Count > 0)
            {
                _controlFlowStack.Peek().ContinueFlag = value;
            }
        }
    }
    
    private class ControlFlowState
    {
        public bool BreakFlag { get; set; }
        public bool ContinueFlag { get; set; }
    }
}
```

### 6.4 ForStatement.Run方法示例

```csharp
public override void Run(VariateManager manager)
{
    manager.AddChildren();
    manager.ControlFlowManager.PushState();
    
    try
    {
        setStatement.Run(manager);
        while (true)
        {
            // 在每次循环迭代开始时重置控制流标志
            manager.ControlFlowManager.ResetCurrentState();
            
            var varExpr = expression.Run(manager);
            bool expr1;
            if (varExpr is BoolLangValue value)
            {
                expr1 = value.Value;
                value.ReturnToPool();
            }
            else
                throw new TypeError(this, "期望布尔类型", $"实际得到了 {varExpr.GetType().Name}");
            
            if (expr1)
            {
                blockStatement.Run(manager);
                
                if (manager.ControlFlowManager.BreakFlag)
                {
                    break;
                }
                
                if (manager.ControlFlowManager.ContinueFlag)
                {
                    statement.Run(manager);
                    continue;
                }
                
                statement.Run(manager);
            }
            else
                break;
        }
    }
    finally
    {
        manager.ControlFlowManager.PopState();
        manager.RemoveChildren();
    }
}
```
